## 1. **数据库操作**

```sql
-- 创建数据库
CREATE DATABASE test_db;

-- 查看数据库
SHOW DATABASES;

-- 使用数据库
USE test_db;

-- 删除数据库
DROP DATABASE test_db;
```

---

## 2. **数据表操作**

```sql
-- 创建表
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    age INT,
    email VARCHAR(100) UNIQUE
);

-- 查看表
SHOW TABLES;

-- 查看表结构
DESC users;

-- 修改表（加字段）
ALTER TABLE users ADD COLUMN phone VARCHAR(20);

-- 修改表（改字段类型）
ALTER TABLE users MODIFY COLUMN age SMALLINT;

-- 删除表
DROP TABLE users;
```

---

## 3. **增删改查（CRUD）**

```sql
-- 插入数据
INSERT INTO users (name, age, email) VALUES ('Alice', 23, 'alice@test.com');

-- 批量插入
INSERT INTO users (name, age, email) VALUES 
('Bob', 25, 'bob@test.com'),
('Charlie', 30, 'charlie@test.com');

-- 查询全部
SELECT * FROM users;

-- 查询指定字段
SELECT name, age FROM users;

-- 带条件查询
SELECT * FROM users WHERE age > 25;

-- 修改数据
UPDATE users SET age = 26 WHERE name = 'Bob';

-- 删除数据
DELETE FROM users WHERE name = 'Charlie';
```

---

## 4. **条件与排序**

```sql
-- 排序
SELECT * FROM users ORDER BY age DESC;

-- 去重
SELECT DISTINCT age FROM users;

-- 限制条数
SELECT * FROM users LIMIT 5;

-- 分页 (第 2 页，每页 5 条)
SELECT * FROM users LIMIT 5 OFFSET 5;
```

---

## 5. **聚合函数**

```sql
-- 统计行数
SELECT COUNT(*) FROM users;

-- 最大/最小/平均值
SELECT MAX(age), MIN(age), AVG(age) FROM users;

-- 分组统计
SELECT age, COUNT(*) FROM users GROUP BY age;

-- 分组后筛选 (HAVING)
SELECT age, COUNT(*) FROM users GROUP BY age HAVING COUNT(*) > 1;
```

## 6. **连接 (JOIN)**

```sql
-- 创建另一张表
CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    product VARCHAR(100),
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 内连接
SELECT users.name, orders.product 
FROM users
JOIN orders ON users.id = orders.user_id;

-- 左连接
SELECT users.name, orders.product 
FROM users
LEFT JOIN orders ON users.id = orders.user_id;

-- 右连接
SELECT users.name, orders.product 
FROM users
RIGHT JOIN orders ON users.id = orders.user_id;
```

## 7. **子查询**

```sql
-- 查询比平均年龄大的用户
SELECT * FROM users WHERE age > (SELECT AVG(age) FROM users);

-- IN 子查询
SELECT name FROM users WHERE id IN (SELECT user_id FROM orders);
```

## 8. **索引**

```sql
-- 创建索引
CREATE INDEX idx_name ON users(name);

-- 查看索引
SHOW INDEX FROM users;

-- 删除索引
DROP INDEX idx_name ON users;
```

## 9. **事务控制**

```sql
-- 开启事务
START TRANSACTION;

-- 插入/更新
INSERT INTO users (name, age, email) VALUES ('David', 28, 'david@test.com');

-- 回滚
ROLLBACK;

-- 提交
COMMIT;
```

## 10. **用户与权限**

```sql
-- 创建用户
CREATE USER 'testuser'@'localhost' IDENTIFIED BY 'password';

-- 授权
GRANT ALL PRIVILEGES ON test_db.* TO 'testuser'@'localhost';

-- 收回权限
REVOKE ALL PRIVILEGES ON test_db.* FROM 'testuser'@'localhost';

-- 查看权限
SHOW GRANTS FOR 'testuser'@'localhost';
```
